function openPhantomDirectly() { const rawUrl = "https://shenouda77.github.io/notbot-app/"; const encodedUrl = encodeURIComponent(rawUrl); // بروتوكول فانتوم المباشر لتشغيل المتصفح الداخلي فوراً const phantomAppUri = "phantom://browse/" + encodedUrl; const phantomUniversal = "https://phantom.app/ul/browse/" + encodedUrl + "?ref=" + encodeURIComponent("https://shenouda77.github.io"); // محاولة الفتح عبر البروتوكول المباشر أولاً window.location.href = phantomAppUri; // في حال عدم الاستجابة خلال ثانية، التوجيه عبر الرابط العام setTimeout(() => { if (window.Telegram && window.Telegram.WebApp) { window.Telegram.WebApp.openLink(phantomUniversal); } else { window.location.href = phantomUniversal; } }, 1000); }
